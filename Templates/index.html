from pytgcalls import PyTgCalls
from pytgcalls.types import AudioPiped

async def play_music(chat_id, song_query):
    # Direct stream URL from your API (Zero downloading needed)
    stream_url = f"http://localhost:5000/download?title={song_query}&type=audio&stream=1"
    
    # Play directly in voice chat via Assistant
    await pytgcalls.join_group_call(
        chat_id,
        AudioPiped(stream_url)
    )
Python Telegram Bot (Download & Send Audio) 
# Download & Send to Telegram User - Made by Yuvi
import requests

def get_audio_file(query):
    api_url = "http://localhost:5000/download"
    res = requests.get(api_url, params={"title": query, "type": "audio"})
    with open("track.mp3", "wb") as f:
        f.write(res.content)
    return "track.mp3"
